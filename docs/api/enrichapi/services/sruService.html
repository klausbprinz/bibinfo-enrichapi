 45    async def fetchRecord(self, identifier: str, idType: str) -> etree._Element | None:
 46        """Handles the HTTP lifecycle and parses raw XML into an lxml tree element."""
 47        cleanId = identifier.strip()
 48        safeId = quote(cleanId, safe="")
 49        
 50        # no double quotes around safeId -> try this now..
 51        query = (
 52            f'alma.barcode={safeId}' 
 53            if idType == "barcode" 
 54            else f'alma.local_control_field_009={safeId}'
 55        )
 56        
 57        url = f"{self.baseUrl}?version=1.2&operation=searchRetrieve&query={query}"
 58        
 59        logger.info(f"[SRU] Initiating primary record fetch | id='{cleanId}' type='{idType}' query='{query}'")
 60        startTime = time.perf_counter()
 61
 62        try:
 63            res = await self.client.get(url)
 64            elapsedMs = round((time.perf_counter() - startTime) * 1000, 2)
 65            res.raise_for_status()
 66            
 67            sruRoot = etree.fromstring(res.content)
 68            recordElems = sruRoot.findall(".//srw:record", namespaces=self.ns)
 69            
 70            if len(recordElems) == 1:
 71                logger.info(f"[SRU] Successfully fetched and matched primary MARC21 record for '{cleanId}' ({elapsedMs}ms)")
 72                return recordElems[0].find("srw:recordData/marc:record", namespaces=self.ns)
 73            elif len(recordElems) == 0:
 74                logger.warning(f"[SRU] No records found for identifier '{cleanId}' (query='{query}') ({elapsedMs}ms)")
 75            else:
 76                logger.warning(f"[SRU] Multiple records ({len(recordElems)}) returned for unique id '{cleanId}' ({elapsedMs}ms)")
 77                
 78            return None
 79
 80        except httpx.TimeoutException:
 81            elapsedMs = round((time.perf_counter() - startTime) * 1000, 2)
 82            logger.error(f"[SRU] Timeout after {elapsedMs}ms connecting to SRU server for query: {query}")
 83            return None
 84        except httpx.HTTPError as exc:
 85            elapsedMs = round((time.perf_counter() - startTime) * 1000, 2)
 86            logger.error(f"[SRU] HTTP status error ({exc.response.status_code if hasattr(exc, 'response') else 'N/A'}) fetching query='{query}' ({elapsedMs}ms)", exc_info=True)
 87            return None
 88        except etree.XMLSyntaxError as exc:

 93    def extractMarc21Metadata(self, marcRecord: etree._Element) -> BasicMarc21MD:
 94        """Transforms raw XML selectors into pydantic object."""
 95        logger.info("[SRU] Parsing raw MARC21 XML into Pydantic data models...")
 96
 97        # extract title
 98        titleData = parse.extractTitleData(marcRecord)
 99        titleModel = Marc21MdTitle(
100            titleMain=titleData[0],
101            titleRemainder=titleData[1],
102            titlePartNumber=titleData[2],
103            titlePartName=titleData[3]
104        )
105        logger.debug(f"[SRU] Extracted title: '{titleModel.titleMain}'")
106
107        # extract main entry (100 / 110 / 111)
108        mainEntryModel = None
109        for tag, nameType in [("100", "person"), ("110", "corporate"), ("111", "conferenceOrEvent")]:
110            field = marcRecord.find(f'marc:datafield[@tag="{tag}"]', namespaces=self.ns)
111            if field is not None:
112                data = parse.extractEntryData(field)
113                if data:
114                    mainEntryModel = Marc21MdMainEntry(nameType=nameType, **data)
115                    logger.info(
116                        f"[SRU] Main entry found (tag {tag}, type={nameType}): '{mainEntryModel.name}' "
117                        f"| GND='{mainEntryModel.gndIdentifier}'"
118                    )
119                    break
120        if not mainEntryModel:
121            logger.debug("[SRU] No main entry (100/110/111) found in record.")
122
123        # extract added entries (700 / 710 / 711)
124        addedEntriesModels: list[Marc21MdAddedEntry] = []
125        tagMap = {"700": "person", "710": "corporate", "711": "conferenceOrEvent"}
126        
127        df7xxList = marcRecord.xpath(
128            'marc:datafield[@tag="700" or @tag="710" or @tag="711"]',
129            namespaces=self.ns
130        )
131        for field in df7xxList:
132            nameType = tagMap.get(field.attrib.get("tag"))
133            data = parse.extractEntryData(field)
134            if data and nameType:
135                addedEntriesModels.append(Marc21MdAddedEntry(nameType=nameType, **data))
136        logger.debug(f"[SRU] Extracted {len(addedEntriesModels)} added entries (700/710/711).")
137
138        # extract physDescriptions
139        physDescriptModels = [
140            Marc21MdPhysDescription(**d) 
141            for d in parse.extractPhysicalDescriptions(marcRecord)
142        ]
143        logger.debug(f"[SRU] Extracted {len(physDescriptModels)} physical descriptions.")
144
145        # extract publicationNotices

147            Marc21MdPublicationNotice(**d) 
148            for d in parse.extractPublicationNotices(marcRecord)
149        ]
150        logger.debug(f"[SRU] Extracted {len(publicationNoticeModels)} publication notices.")
151
152        # extract classifications
153        classificationModels = [
154            Marc21MdClassificationNumber(**d)

156        ]
157        logger.debug(f"[SRU] Extracted {len(classificationModels)} classification entries.")
158
159        # extract identifier

161            Marc21MdIdentifier(**d) 
162            for d in parse.extractIdentifier(marcRecord)
163        ]

172
173        # scalar / list extractions
174        langCodes = parse.extractLanguageCodes(marcRecord)
175        subjectHeadings = parse.extractSubjectHeadings(marcRecord)
176        logger.debug(f"[SRU] Metadata summary | Languages: {langCodes} | Subject Headings count: {len(subjectHeadings)}")

206    async def fetchAdditionalRecords(self, marcData: BasicMarc21MD, data) -> dict[str, Any]:
207        """
208        Reuses structured data already parsed by extractMarc21Metadata 
209        to coordinate dynamic secondary SRU lookups based on user flags.
210        """
211        tasks = []
212        taskTypes = []
213        finalRecords = []
214
215        # extract primaryBib identifier (controlfield 009)
216        baseAc = None
217        if marcData.identifier:
218            for idObj in marcData.identifier:
219                if idObj.idType == "primaryBib" and idObj.value:
220                    baseAc = idObj.value
221                    break
222
223        logger.info(f"[SRU] Preparing subsidiary searches for base record AC='{baseAc}'")
224
225        # author search
226        if data.fetchSimilarByAuthor:
227            if marcData.mainEntry and marcData.mainEntry.name:
228                authorName = marcData.mainEntry.name
229                logger.info(f"[SRU] Queuing subsidiary author search for '{authorName}'")
230                tasks.append(
231                    self._executeSubsidiarySRU(
232                        f'alma.creator="{authorName}"', 
233                        baseIdentifier=baseAc, 
234                        maxRecs=data.maxRecs
235                    )
236                )
237                taskTypes.append(("author", authorName))
238            else:
239                # explicit empty return
240                logger.warning("[SRU] Author search requested, but no main entry author was found in MARC data.")
241                finalRecords.append(
242                    AdditionalRecsByAuthor(
243                        searchType="author",
244                        name="",
245                        maxRecs=data.maxRecs,
246                        additionalRecs=[]
247                    )
248                )
249
250        # subject headings search
251        if data.fetchSimilarBySubject:
252            if marcData.subjectHeadings:
253                queryStr = " AND ".join([f'alma.subject="{s}"' for s in marcData.subjectHeadings])
254                logger.info(f"[SRU] Queuing subsidiary subject search with {len(marcData.subjectHeadings)} terms")
255                tasks.append(
256                    self._executeSubsidiarySRU(
257                        queryStr, 
258                        baseIdentifier=baseAc, 
259                        maxRecs=data.maxRecs
260                    )
261                )
262                taskTypes.append(("subject", marcData.subjectHeadings))
263            else:
264                logger.warning("[SRU] Subject search requested, but no subject headings were found in MARC data.")
265                finalRecords.append(
266                    AdditionalRecsBySubjectHeadings(
267                        searchType="subjectHeadings",
268                        subjectHeadings=[],
269                        maxRecs=data.maxRecs,
270                        additionalRecs=[]
271                    )
272                )
273
274        # classifications search
275        if data.fetchSimilarByClassification:
276            targetClassifications = []
277            indexName = "alma.other_class_number"
278
279            if marcData.classifications:
280                # group available classification objects by scheme type (lowercase)
281                schemeMap: dict[str, list] = {}
282                for c in marcData.classifications:
283                    if not c.classificationNumber:
284                        continue
285                    ctype = (c.classificationType or "").lower().strip()
286                    schemeMap.setdefault(ctype, []).append(c)
287
288                # prioritization hierarchy: BKL -> RVK -> DDC -> Fallback to all others
289                chosenSchemeObjs = []
290                if "bkl" in schemeMap:
291                    chosenSchemeObjs = schemeMap["bkl"]
292                    indexName = "alma.other_class_number"
293                elif "rvk" in schemeMap:
294                    chosenSchemeObjs = schemeMap["rvk"]
295                    indexName = "alma.other_class_number"
296                elif "ddc" in schemeMap:
297                    chosenSchemeObjs = schemeMap["ddc"]
298                    indexName = "alma.dewey_decimal_class_number"
299                else:
300                    # flatten remaining non-empty schemes if none of top 3 are found
301                    chosenSchemeObjs = [c for cList in schemeMap.values() for c in cList]
302                    indexName = "alma.other_class_number"
303
304                targetClassifications = chosenSchemeObjs
305
306            if targetClassifications:
307                # extract numbers and deduplicate preserving order
308                classNumbers = list(dict.fromkeys([c.classificationNumber for c in targetClassifications]))
309                
310                # build SRU clauses joined by AND for precision matching
311                queryClauses = [f'{indexName}="{num}"' for num in classNumbers]
312                queryStr = " AND ".join(queryClauses)
313
314                logger.info(f"[SRU] Queuing classification search using index '{indexName}' with values: {classNumbers}")
315                tasks.append(
316                    self._executeSubsidiarySRU(
317                        queryStr, 
318                        baseIdentifier=baseAc, 
319                        maxRecs=data.maxRecs
320                    )
321                )
322                taskTypes.append(("classification", targetClassifications))
323            else:
324                logger.warning("[SRU] Classification search requested, but no valid classification numbers were found in MARC data.")
325                finalRecords.append(
326                    AdditionalRecsByClassification(
327                        searchType="classification",
328                        classifications=[],
329                        maxRecs=data.maxRecs,
330                        additionalRecs=[]
331                    )
332                )
333
334        # execute dynamic tasks concurrently
335        if tasks:
336            logger.info(f"[SRU] Executing {len(tasks)} subsidiary search task(s) concurrently...")
337            results = await asyncio.gather(*tasks, return_exceptions=True)
338
339            for (category, criteria), res in zip(taskTypes, results):
340                if isinstance(res, Exception):
341                    logger.error(f"[SRU] Subsidiary search for category '{category}' failed: {res}", exc_info=True)
342                    recs = []
343                else:
344                    recs = [AdditionalRec(**r) if isinstance(r, dict) else r for r in res]
345                    logger.info(f"[SRU] Subsidiary '{category}' search returned {len(recs)} record(s)")
346
347                if category == "author":
348                    finalRecords.append(
349                        AdditionalRecsByAuthor(
350                            searchType="author",
351                            name=criteria,
352                            maxRecs=data.maxRecs,
353                            additionalRecs=recs
354                        )
355                    )
356                elif category == "subject":
357                    finalRecords.append(
358                        AdditionalRecsBySubjectHeadings(
359                            searchType="subjectHeadings",
360                            subjectHeadings=criteria,
361                            maxRecs=data.maxRecs,
362                            additionalRecs=recs
363                        )
364                    )
365                elif category == "classification":
366                    finalRecords.append(
367                        AdditionalRecsByClassification(
368                            searchType="classification",
369                            classifications=marcData.classifications,
370                            maxRecs=data.maxRecs,
371                            additionalRecs=recs
372                        )
373                    )
374
375        return {"records": finalRecords}

378    async def _executeSubsidiarySRU(
379        self, 
380        queryString: str, 
381        baseIdentifier: str | None = None, 
382        maxRecs: int = 5
383    ) -> list[dict]:
384        """Helper network worker to fetch and parse response records from subsidiary queries."""
385        
386        # always fetch 1 extra record to account for potential base record inclusion
387        fetchCount = maxRecs + 1 if baseIdentifier else maxRecs
388
389        params = {
390            "version": "1.2",
391            "operation": "searchRetrieve",
392            "query": queryString,
393            "maximumRecords": str(fetchCount)
394        }
395
396        startTime = time.perf_counter()
397        try:
398            res = await self.client.get(self.baseUrl, params=params)
399            elapsedMs = round((time.perf_counter() - startTime) * 1000, 2)
400            res.raise_for_status()
401
402            sruRoot = etree.fromstring(res.content)
403            recordDataElems = sruRoot.findall(".//srw:recordData/marc:record", namespaces=self.ns)
404
405            extractedRecords = []
406            baseIdClean = baseIdentifier.strip() if baseIdentifier else None
407
408            for recordElem in recordDataElems:
409                recDict = parse.extractAdditionalRecData(recordElem)
410                
411                # guardrail 1: filter out base record if matched
412                if baseIdClean and recDict.get("ac") == baseIdClean:
413                    logger.debug(f"[SRU] Base record '{baseIdClean}' filtered from subsidiary search results")
414                    continue
415
416                extractedRecords.append(recDict)
417
418            finalBatch = extractedRecords[:maxRecs]
419            logger.info(f"[SRU] Subsidiary query='{queryString}' returned {len(recordDataElems)} raw items -> {len(finalBatch)} filtered items ({elapsedMs}ms)")
420            
421            # guardrail 2: enforce strict maxRecs upper limit regardless of whether 0 or 1 item was filtered
422            return finalBatch
423
424        except Exception as e:
425            elapsedMs = round((time.perf_counter() - startTime) * 1000, 2)
426            logger.error(f"[SRU] Error during subsidiary query='{queryString}' ({elapsedMs}ms): {e}", exc_info=True)
427            # re-raise so asyncio.gather(return_exceptions=True) catches it cleanly per task
428            raise e

383        maxRecs: int = 5
384    ) -> list[dict]:
385        """Helper network worker to fetch and parse response records from subsidiary queries."""
386        
387        # always fetch 1 extra record to account for potential base record inclusion
388        fetchCount = maxRecs + 1 if baseIdentifier else maxRecs
389
390        params = {
391            "version": "1.2",
392            "operation": "searchRetrieve",
393            "query": queryString,
394            "maximumRecords": str(fetchCount)
395        }
396
397        startTime = time.perf_counter()
398        try:
399            res = await self.client.get(self.baseUrl, params=params)
400            elapsedMs = round((time.perf_counter() - startTime) * 1000, 2)
401            res.raise_for_status()
402
403            sruRoot = etree.fromstring(res.content)
404            recordDataElems = sruRoot.findall(".//srw:recordData/marc:record", namespaces=self.ns)
405
406            extractedRecords = []
407            baseIdClean = baseIdentifier.strip() if baseIdentifier else None
408
409            for recordElem in recordDataElems:

411                
412                # guardrail 1: filter out base record if matched
413                if baseIdClean and recDict.get("ac") == baseIdClean:
414                    logger.debug(f"[SRU] Base record '{baseIdClean}' filtered from subsidiary search results")
415                    continue
416
417                extractedRecords.append(recDict)
418
419            finalBatch = extractedRecords[:maxRecs]
420            logger.info(f"[SRU] Subsidiary query='{queryString}' returned {len(recordDataElems)} raw items -> {len(finalBatch)} filtered items ({elapsedMs}ms)")
421            
422            # guardrail 2: enforce strict maxRecs upper limit regardless of whether 0 or 1 item was filtered
423            return finalBatch
424
425        except Exception as e:

427            logger.error(f"[SRU] Error during subsidiary query='{queryString}' ({elapsedMs}ms): {e}", exc_info=True)
