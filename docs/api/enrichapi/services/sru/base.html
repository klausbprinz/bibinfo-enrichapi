 88        cleanId = identifier.strip()
 89        
 90        # normalize legacy or alternate aliases to the universal 'bibId' standard
 91        normalizedIdType = "bibId" if idType in ("bibId", "ac", "ppn") else idType

 94        url = f"{self.baseUrl}?version=1.2&operation=searchRetrieve&query={query}&maximumRecords=50"
 95        if self.recordSchema:
 96            url += f"&recordSchema={self.recordSchema}"

 98        logger.info(f"[SRU] Initiating primary record fetch | id='{cleanId}' type='{normalizedIdType}' query='{query}'")
 99        startTime = time.perf_counter()
100
101        warnings: list[str] = []
102
103        try:
104            res = await self.client.get(url)
105            elapsedMs = round((time.perf_counter() - startTime) * 1000, 2)

113            if normalizedIdType == "bibId":

116                    return recordElems[0].find("srw:recordData/marc:record", namespaces=self.ns), warnings
117                elif count == 0:
118                    logger.warning(f"[SRU] No records found for system identifier '{cleanId}' (query='{query}') ({elapsedMs}ms)")
119                else:

121                return None, warnings
122
123            # rule 2: pragmatic barcodes (auto-selects first match on collision with warning)
124            elif normalizedIdType == "barcode":
125                if count == 1:
126                    logger.info(f"[SRU] Successfully fetched and matched primary MARC21 record for barcode '{cleanId}' ({elapsedMs}ms)")
127                    return recordElems[0].find("srw:recordData/marc:record", namespaces=self.ns), warnings
128                elif count == 0:

130                    return None, warnings

133                    logger.warning(f"[SRU] {warningMsg} ({elapsedMs}ms)")
134                    warnings.append(warningMsg)
135                    return recordElems[0].find("srw:recordData/marc:record", namespaces=self.ns), warnings

144                    return None, warnings

162                                    if entryData and entryData.get("name"):
163                                        authorName = entryData.get("name")
164                                        break
165                            
166                            pubNotices = parse.extractPublicationNotices(marcRec)
167                            publishersVal = pubNotices[0].get("names") if pubNotices else None

169
170                            candidates.append(

173                                    title=fullTitle,
174                                    author=authorName,
175                                    publishers=publishersVal,
176                                    dates=yearsVal
177                                )
178                            )
179

181                        isbn=cleanId,

192            return None, warnings
193        except httpx.HTTPError as exc:

195            logger.error(f"[SRU] HTTP status error fetching query='{query}' ({elapsedMs}ms)", exc_info=True)
196            return None, warnings
197        except etree.XMLSyntaxError as exc:

205    def extractMarc21Metadata(self, marcRecord: etree._Element) -> BasicMarc21MD:
206        """Transforms raw MARC21 XML selectors into structured Pydantic models."""
207        logger.info("[SRU] Parsing raw MARC21 XML into Pydantic data models...")
208
209        titleData = parse.extractTitleData(marcRecord)
210        titleModel = Marc21MdTitle(
211            titleMain=titleData[0],
212            titleRemainder=titleData[1],
213            titlePartNumber=titleData[2],
214            titlePartName=titleData[3]
215        )
216
217        mainEntryModel = None
218        for tag, nameType in [("100", "person"), ("110", "corporate"), ("111", "conferenceOrEvent")]:
219            field = marcRecord.find(f'marc:datafield[@tag="{tag}"]', namespaces=self.ns)
220            if field is not None:
221                data = parse.extractEntryData(field)
222                if data:
223                    mainEntryModel = Marc21MdMainEntry(nameType=nameType, **data)
224                    break
225
226        addedEntriesModels: list[Marc21MdAddedEntry] = []
227        tagMap = {"700": "person", "710": "corporate", "711": "conferenceOrEvent"}
228        df7xxList = marcRecord.xpath('marc:datafield[@tag="700" or @tag="710" or @tag="711"]', namespaces=self.ns)
229        for field in df7xxList:
230            nameType = tagMap.get(field.attrib.get("tag"))
231            data = parse.extractEntryData(field)
232            if data and nameType:
233                addedEntriesModels.append(Marc21MdAddedEntry(nameType=nameType, **data))
234
235        physDescriptModels = [Marc21MdPhysDescription(**d) for d in parse.extractPhysicalDescriptions(marcRecord)]
236        publicationNoticeModels = [Marc21MdPublicationNotice(**d) for d in parse.extractPublicationNotices(marcRecord)]
237        classificationModels = [Marc21MdClassificationNumber(**d) for d in parse.extractClassificationNumbers(marcRecord)]
238        identifierModels = [Marc21MdIdentifier(**d) for d in parse.extractIdentifier(marcRecord)]
239        electronicLocationModels = [Marc21MdElectronicLocation(**d) for d in parse.extractElectronicLocations(marcRecord)]
240        holdingModels = [Marc21MdHoldingInfos(**d) for d in parse.extractHoldingInfos(marcRecord)]
241
242        langCodes = parse.extractLanguageCodes(marcRecord)
243        subjectHeadings = parse.extractSubjectHeadings(marcRecord)
244
245        return BasicMarc21MD(
246            title=titleModel,
247            mainEntry=mainEntryModel,
248            addedEntries=addedEntriesModels,
249            languageCodes=langCodes,
250            languageCodesOriginal=parse.extractLanguageCodesOriginal(marcRecord),
251            publicationCountryCodes=parse.extractPublicationCountryCodes(marcRecord),
252            edition=parse.extractEdition(marcRecord),
253            physicalDescriptions=physDescriptModels,
254            publicationNotices=publicationNoticeModels,
255            genreForms=parse.extractGenreForms(marcRecord),
256            subjectHeadings=subjectHeadings,
257            classifications=classificationModels,
258            bibMaterialType=parse.extractBibMaterialType(marcRecord),
259            bibResourceType=parse.extractBibResourceType(marcRecord),
260            electronicLocations=electronicLocationModels,
261            abstracts=parse.extractAbstracts(marcRecord),
262            identifier=identifierModels,
263            holdingInfos=holdingModels
264        )

270    async def fetchAdditionalRecords(self, marcData: BasicMarc21MD, data) -> dict[str, Any]:
271        """Coordinates dynamic secondary SRU lookups concurrently using institutional query builders."""
272        tasks = []
273        taskTypes = []
274        finalRecords = []
275
276        baseBibId = self.extractBaseBibId(marcData)
277        logger.info(f"[SRU] Preparing subsidiary searches for base record ID='{baseBibId}'")
278
279        # 1. person-based similarity query
280        if data.fetchSimilarByPerson:
281            if marcData.mainEntry and marcData.mainEntry.name:
282                personName = marcData.mainEntry.name
283                queryStr = self.buildPersonQuery(personName)
284                tasks.append(self._executeSubsidiarySRU(queryStr, baseIdentifier=baseBibId, maxRecs=data.maxRecs))
285                taskTypes.append(("person", personName))
286            else:
287                finalRecords.append(AdditionalRecsByPerson(searchType="person", name="", maxRecs=data.maxRecs, additionalRecs=[]))
288
289        # 2. subject-headings-based similarity query
290        if data.fetchSimilarBySubject:
291            if marcData.subjectHeadings:
292                queryStr = self.buildSubjectQuery(marcData.subjectHeadings)
293                tasks.append(self._executeSubsidiarySRU(queryStr, baseIdentifier=baseBibId, maxRecs=data.maxRecs))
294                taskTypes.append(("subject", marcData.subjectHeadings))
295            else:
296                finalRecords.append(AdditionalRecsBySubjectHeadings(searchType="subjectHeadings", subjectHeadings=[], maxRecs=data.maxRecs, additionalRecs=[]))
297
298        # 3. classification-based similarity query
299        if data.fetchSimilarByClassification:
300            if marcData.classifications:
301                queryStr = self.buildClassificationQuery(marcData.classifications)
302                if queryStr:
303                    tasks.append(self._executeSubsidiarySRU(queryStr, baseIdentifier=baseBibId, maxRecs=data.maxRecs))
304                    taskTypes.append(("classification", marcData.classifications))
305                else:
306                    finalRecords.append(AdditionalRecsByClassification(searchType="classification", classifications=[], maxRecs=data.maxRecs, additionalRecs=[]))
307            else:
308                finalRecords.append(AdditionalRecsByClassification(searchType="classification", classifications=[], maxRecs=data.maxRecs, additionalRecs=[]))
309
310        # execute dynamic subsidiary tasks concurrently via asyncio.gather
311        if tasks:
312            results = await asyncio.gather(*tasks, return_exceptions=True)
313            for (category, criteria), res in zip(taskTypes, results):
314                if isinstance(res, Exception):
315                    recs = []
316                else:
317                    recs = [AdditionalRec(**r) if isinstance(r, dict) else r for r in res]
318
319                if category == "person":
320                    finalRecords.append(AdditionalRecsByPerson(searchType="person", name=criteria, maxRecs=data.maxRecs, additionalRecs=recs))
321                elif category == "subject":
322                    finalRecords.append(AdditionalRecsBySubjectHeadings(searchType="subjectHeadings", subjectHeadings=criteria, maxRecs=data.maxRecs, additionalRecs=recs))
323                elif category == "classification":
324                    finalRecords.append(AdditionalRecsByClassification(searchType="classification", classifications=criteria, maxRecs=data.maxRecs, additionalRecs=recs))
325
326        return {"records": finalRecords}

328    async def _executeSubsidiarySRU(self, queryString: str, baseIdentifier: str | None = None, maxRecs: int = 5) -> list[dict]:
329        """Worker helper to fetch subsidiary records and filter out the base record match."""
330        # fetch 1 extra record to account for potential base record self-inclusion
331        fetchCount = maxRecs + 1 if baseIdentifier else maxRecs
332        params = {"version": "1.2", "operation": "searchRetrieve", "query": queryString, "maximumRecords": str(fetchCount)}
333        if self.recordSchema:
334            params["recordSchema"] = self.recordSchema
335        
336        startTime = time.perf_counter()
337        try:
338            res = await self.client.get(self.baseUrl, params=params, timeout=8.0)
339            elapsedMs = round((time.perf_counter() - startTime) * 1000, 2)
340            res.raise_for_status()
341
342            sruRoot = etree.fromstring(res.content)
343            recordDataElems = sruRoot.findall(".//srw:recordData/marc:record", namespaces=self.ns)
344
345            extractedRecords = []
346            baseIdClean = baseIdentifier.strip() if baseIdentifier else None
347
348            for recordElem in recordDataElems:
349                recDict = parse.extractAdditionalRecData(recordElem)
350                # guardrail: filter out the original source record from similar lists
351                if baseIdClean and recDict.get("bibId") == baseIdClean:
352                    continue
353                extractedRecords.append(recDict)
354
355            return extractedRecords[:maxRecs]
356        except Exception as e:
357            raise e

329    async def _executeSubsidiarySRU(self, queryString: str, baseIdentifier: str | None = None, maxRecs: int = 5) -> list[dict]:
330        """Worker helper to fetch subsidiary records and filter out the base record match."""
331        # fetch 1 extra record to account for potential base record self-inclusion
332        fetchCount = maxRecs + 1 if baseIdentifier else maxRecs
333        params = {"version": "1.2", "operation": "searchRetrieve", "query": queryString, "maximumRecords": str(fetchCount)}
334        if self.recordSchema:
335            params["recordSchema"] = self.recordSchema
336        
337        startTime = time.perf_counter()
338        try:
339            res = await self.client.get(self.baseUrl, params=params, timeout=8.0)
340            elapsedMs = round((time.perf_counter() - startTime) * 1000, 2)
341            res.raise_for_status()
342
343            sruRoot = etree.fromstring(res.content)
344            recordDataElems = sruRoot.findall(".//srw:recordData/marc:record", namespaces=self.ns)
345
346            extractedRecords = []
347            baseIdClean = baseIdentifier.strip() if baseIdentifier else None
348
349            for recordElem in recordDataElems:
350                recDict = parse.extractAdditionalRecData(recordElem)
351                # guardrail: filter out the original source record from similar lists
352                if baseIdClean and recDict.get("bibId") == baseIdClean:
353                    continue
354                extractedRecords.append(recDict)
355
356            return extractedRecords[:maxRecs]
357        except Exception as e:
358            raise e
