 96def resolveWikidataType(v: Any) -> str:
 97    """
 98    Discriminator reading explicit entity type passed down from MARC21 nameType.
 99    Falls back to checking P31 (instanceOf) or 'default' if omitted.
100    """
101    targetType = None
102
103    if isinstance(v, BaseModel):
104        targetType = getattr(v, "wikidataType", None)
105    elif isinstance(v, dict):
106        targetType = v.get("wikidataType") or v.get("entityType")
107
108    # primary route: explicit type passed from MARC21 nameType
109    if targetType in ("person", "corporate", "conferenceOrEvent"):
110        return targetType
111
112    # fallback route: inspect P31 if no explicit type provided
113    p31List = []
114    if isinstance(v, BaseModel):
115        p31List = getattr(v, "instanceOf", []) or getattr(v, "P31", [])
116    elif isinstance(v, dict):
117        p31List = v.get("P31", []) or v.get("instanceOf", [])
118
119    if isinstance(p31List, str):
120        p31List = [p31List]
121
122    p31Set = {str(item).lower().strip() for item in p31List}
123
124    personMarkers = {"q5", "mensch", "human"}
125    corporateMarkers = {
126        "q43229", "organisation", "organization", "unternehmen", 
127        "business", "bibliothek", "library", "hochschule", "university"
128    }
129    eventMarkers = {"q1656682", "ereignis", "event", "konferenz", "conference"}
130
131    if p31Set.intersection(personMarkers):
132        return "person"
133    if p31Set.intersection(corporateMarkers):
134        return "corporate"
135    if p31Set.intersection(eventMarkers):
