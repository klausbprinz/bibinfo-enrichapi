141    @computed_field
142    @property
143    def wikidataId(self) -> str | None:
144        """
145        Inspects sameAs items for collectionName=='Wikidata' or wikidata.org URL 
146        and returns the plain Q-ID (e.g. 'Q9312').
147        """
148        for item in self.sameAs:
149            # match explicitly by collection name or URL pattern
150            isWikidata = (
151                item.collectionName == "Wikidata" 
152                or (item.idURL and "wikidata.org/entity/" in item.idURL)
153            )
154            if isWikidata and item.idURL:
155                # extract trailing Q-ID from URL (e.g. 'http://www.wikidata.org/entity/Q9312' -> 'Q9312')
156                return item.idURL.rstrip("/").rsplit("/", 1)[-1]
157        return None

259# discriminator function to determine the model dynamically
260def resolveGndType(v: Any) -> str:
261    # extract types list from dict or model instance
262    types = []
263    if isinstance(v, BaseModel):
264        types = getattr(v, "entityTypes", []) or getattr(v, "type", [])
265    elif isinstance(v, dict):
266        types = v.get("type", []) or v.get("entityTypes", [])
267    
268    if isinstance(types, str):
269        types = [types]
270
271    # inspect type array for matching entity tag
272    if "Person" in types:
273        return "person"
274    if "CorporateBody" in types:
275        return "corporate"
276    if any(t in types for t in ("Event", "ConferenceOrEvent")):
277        return "conferenceOrEvent"
278
279    # fallback tag for Geografikum, Sachbegriff, Werk, etc.
280    return "default"
